        // Fetch Zones from DB
        async function fetchZonesFromDB() {
            try {
                const zonesSet = new Set();
                
                // Fetch from "zonas" collection if it exists
                const zonasSnap = await getDocs(collection(db, "zonas"));
                zonasSnap.forEach(docSnap => {
                    const data = docSnap.data();
                    if (data.nombre) zonesSet.add(data.nombre);
                    else if (typeof data === 'string') zonesSet.add(data);
                });

                // Fetch distinct zones from existing "agencias" collection
                const agenciasSnap = await getDocs(collection(db, "agencias"));
                agenciasSnap.forEach(docSnap => {
                    const data = docSnap.data();
                    if (data.zona) zonesSet.add(data.zona);
                });

                existingZones = Array.from(zonesSet).sort();
                
                // Fallback default zones if DB has none yet
                if (existingZones.length === 0) {
                    existingZones = ["Zona Norte", "Zona Sur", "Zona Este", "Zona Oeste", "Zona Metropolitana"];
                }

                populateZoneDropdowns(existingZones);

            } catch (error) {
                console.error("Error al cargar zonas:", error);
                // Fallback on error
                existingZones = ["Zona Norte", "Zona Sur", "Zona Este", "Zona Oeste", "Zona Metropolitana"];
                populateZoneDropdowns(existingZones);
            }
        }

                    Swal.fire("Atención", "Escribe o pega los datos en el área de texto.", "warning");
                    return;
                }
                processTextareaContent(rawText, selectedZone);
            }
        });

        function processExcelFile(file, globalZone) {
            const reader = new FileReader();
            reader.onload = (e) => {
                const data = new Uint8Array(e.target.result);
                const workbook = XLSX.read(data, { type: "array" });
                const worksheet = workbook.Sheets[workbook.SheetNames[0]];
                const rawData = XLSX.utils.sheet_to_json(worksheet, { defval: "" });

                parsedAgencies = rawData.map((row) => ({
                    idReal: String(row["idReal"] || row["ID Real"] || row["ID"] || "").trim(),
                    zona: globalZone || String(row["zona"] || row["Zona"] || "").trim() || "General",
                    direccion: String(row["direccion"] || row["Direccion"] || row["Dirección"] || "").trim(),
                    lat: parseFloat(row["lat"] || row["Latitud"] || 0),
                    lng: parseFloat(row["lng"] || row["Longitud"] || 0)
                })).filter(a => a.idReal !== "");

                validateAndShowPreview();
            };
            reader.readAsArrayBuffer(file);
        }

        function processTextareaContent(text, globalZone) {
            const lines = text.split("\n");
            parsedAgencies = [];

            lines.forEach(line => {
                if (!line.trim()) return;
                const delimiter = line.includes("\t") ? "\t" : ",";
                const parts = line.split(delimiter).map(p => p.trim());

                if (parts.length > 0 && parts[0] !== "") {
                    if (parts[0].toLowerCase() === "idreal" || parts[0].toLowerCase() === "id") return;

                    parsedAgencies.push({
                        idReal: String(parts[0] || "").trim(),
                        zona: globalZone || String(parts[1] && isNaN(parts[1]) ? parts[1] : "General").trim(),
                        direccion: parts.length >= 4 ? String(parts[1] || "") : String(parts[1] || ""),
                        lat: parseFloat(parts[2] || 0),
                        lng: parseFloat(parts[3] || 0)
                    });
                }
            });

            validateAndShowPreview();
        }

        function validateAndShowPreview() {
            if (parsedAgencies.length === 0) {
                Swal.fire("Sin Registros", "No se encontraron filas válidas con la columna 'idReal'.", "error");
                btnSaveAgencies.disabled = true;
                previewCard.classList.add("hidden");
                return;
            }

            renderPreviewTable(parsedAgencies);
            btnSaveAgencies.disabled = false;
            
            Swal.fire({
                title: "Previsualización Lista",
                text: `Se cargaron ${parsedAgencies.length} registros correctamente.`,
                icon: "success",
                timer: 2000,
                showConfirmButton: false
            });
        }

        function renderPreviewTable(data) {
            previewTableBody.innerHTML = "";
            recordCountBadge.textContent = `${data.length} registros`;

            data.forEach((item, index) => {
                const tr = document.createElement("tr");
                tr.className = "hover:bg-slate-50 transition-colors";
                tr.innerHTML = `
                    <td class="py-3 px-4 font-mono text-slate-400">${index + 1}</td>
                    <td class="py-3 px-4"><span class="px-2 py-1 rounded bg-sky-50 text-sky-700 font-mono font-bold">${item.idReal}</span></td>
                    <td class="py-3 px-4"><span class="px-2 py-0.5 rounded-full bg-slate-100 text-slate-600 font-medium">${item.zona}</span></td>
                    <td class="py-3 px-4 font-medium">${item.direccion || '-'}</td>
                    <td class="py-3 px-4 font-mono text-slate-500">${item.lat || 0}</td>
                    <td class="py-3 px-4 font-mono text-slate-500">${item.lng || 0}</td>
                `;
                previewTableBody.appendChild(tr);
            });

            previewCard.classList.remove("hidden");
        }

        // Save Data to Firestore with Auto-generated ID
        btnSaveAgencies.addEventListener("click", async () => {
            if (parsedAgencies.length === 0) return;

            const confirmResult = await Swal.fire({
                title: "¿Guardar Agencias?",
                text: `Se procesarán ${parsedAgencies.length} registros en Firestore.`,
                icon: "question",
                showCancelButton: true,
                confirmButtonColor: "#0284c7",
                cancelButtonColor: "#64748b",
                confirmButtonText: "Sí, guardar ahora",
                cancelButtonText: "Cancelar"
            });

            if (!confirmResult.isConfirmed) return;

            btnSaveAgencies.disabled = true;
            btnProcessData.disabled = true;
            progressContainer.classList.remove("hidden");
            updateProgress(0);

            const strategy = importStrategy.value;
            const agenciasRef = collection(db, "agencias");
            
            let processedCount = 0;
            let batch = writeBatch(db);
            let operationCount = 0;

            try {
                for (let i = 0; i < parsedAgencies.length; i++) {
                    const agency = parsedAgencies[i];

                    // Check if document exists by idReal field
                    const q = query(agenciasRef, where("idReal", "==", agency.idReal));
                    const querySnapshot = await getDocs(q);

                    if (!querySnapshot.empty) {
                        if (strategy === "skip") {
                            processedCount++;
                            updateProgress(Math.round((processedCount / parsedAgencies.length) * 100));
                            continue;
                        } else if (strategy === "upsert") {
                            const existingDocId = querySnapshot.docs[0].id;
                            const existingDocRef = doc(db, "agencias", existingDocId);

                            batch.update(existingDocRef, {
                                zona: agency.zona,
                                direccion: agency.direccion,
                                lat: agency.lat,
                                lng: agency.lng,
                                updatedAt: new Date()
                            });
                            operationCount++;
                        }
                    } else {
                        // Create NEW Document with AUTO-GENERATED ID
                        const newDocRef = doc(agenciasRef);

                        batch.set(newDocRef, {
                            idReal: agency.idReal,
                            zona: agency.zona,
                            direccion: agency.direccion,
                            lat: agency.lat,
                            lng: agency.lng,
                            createdAt: new Date()
                        });
                        operationCount++;
                    }

                    processedCount++;
                    updateProgress(Math.round((processedCount / parsedAgencies.length) * 100));

                    if (operationCount === 450) {
                        await batch.commit();
                        batch = writeBatch(db);
                        operationCount = 0;
                    }
                }

                if (operationCount > 0) {
                    await batch.commit();
                }

                Swal.fire({
                    title: "¡Éxito!",
                    text: "Las agencias se guardaron correctamente en Firestore con IDs autogenerados.",
                    icon: "success",
                    confirmButtonColor: "#0284c7"
                });

                // Refresh zones list
                fetchZonesFromDB();

            } catch (error) {
                console.error("Firestore Error:", error);
                Swal.fire("Error", "Error al procesar la carga: " + error.message, "error");
            } finally {
                btnSaveAgencies.disabled = false;
                btnProcessData.disabled = false;
            }
        });

        function updateProgress(percentage) {
            progressBar.style.width = `${percentage}%`;
            progressText.textContent = `${percentage}%`;
        }
    </script>

            '#6366f1', '#10b981', '#f59e0b', '#ec4899', '#8b5cf6',

        function initMap() {
            map = L.map('map').setView([18.7357, -70.1627], 8);

            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                maxZoom: 19,
                attribution: '&copy; OpenStreetMap'
            }).addTo(map);

            markersGroup = L.layerGroup().addTo(map);
        }

        onAuthStateChanged(auth, async (user) => {
            if (!user) {
                window.location.href = "login.html";
                return;
            }
            document.getElementById("userEmail").innerText = user.email;
            initMap();
            await loadData();
            loadingOverlay.classList.add("hidden");

        async function loadData() {
            try {
                const [zonesSnap, agenciasSnap] = await Promise.all([
                    getDocs(collection(db, "zonas")),
                    getDocs(collection(db, "agencias"))
                ]);

                const zonesSet = new Set();

                zonesSnap.forEach(d => {
                    const z = d.id.toUpperCase().trim();
                    if (z) zonesSet.add(z);
                });

                allAgencies = [];
                agenciasSnap.forEach(docSnap => {
                    const data = docSnap.data();
                    const agencyZone = data.zona ? data.zona.toUpperCase().trim() : 'SIN ZONA';
                    if (agencyZone) zonesSet.add(agencyZone);

                    // Solo asigna ID real si existe explícitamente en el documento
                    const realId = data.idReal || data.idreal || data.codigo || null;

                    allAgencies.push({
                        docId: docSnap.id,
                        idreal: realId,
                        nombre: data.nombre || (realId ? realId : 'Agencia sin nombre'),
                        zona: agencyZone,
                        lat: data.latitud || data.lat || null,
                        lng: data.longitud || data.lng || null,
                        estado: data.estado || 'Activa',
                        direccion: data.direccion || ''
                    });
                });

                const sortedZones = Array.from(zonesSet).sort();
                zoneFilter.innerHTML = `<option value="ALL">Todas las Zonas (${sortedZones.length})</option>`;
                editZona.innerHTML = "";

                zoneColorMap = {};
                sortedZones.forEach((zName, index) => {
                    zoneColorMap[zName] = ZONE_COLORS[index % ZONE_COLORS.length];

                    const opt = document.createElement("option");
                    opt.value = zName;
                    opt.textContent = zName;
                    zoneFilter.appendChild(opt);

                    const optEdit = document.createElement("option");
                    optEdit.value = zName;
                    optEdit.textContent = zName;
                    editZona.appendChild(optEdit);
                });

                renderAgenciesOnMap(allAgencies);

            } catch (err) {
                console.error("Error al cargar datos:", err);
                Toastify({ text: "Error al sincronizar con Firestore", style: { background: "#f43f5e" } }).showToast();
            }
        }

        function renderAgenciesOnMap(agencies) {
            markersGroup.clearLayers();
            agencyListContainer.innerHTML = "";

            const validAgencies = agencies.filter(a => a.lat && a.lng);
            agencyCount.innerText = validAgencies.length;

            const bounds = [];

            validAgencies.forEach(agency => {
                const lat = parseFloat(agency.lat);
                const lng = parseFloat(agency.lng);

                if (isNaN(lat) || isNaN(lng)) return;

                bounds.push([lat, lng]);

                const color = zoneColorMap[agency.zona] || '#6366f1';

                const marker = L.circleMarker([lat, lng], {
                    radius: 8,
                    fillColor: color,
                    color: '#ffffff',
                    weight: 2,
                    opacity: 1,
                    fillOpacity: 0.95
                });

                const displayIdPopup = agency.idreal ? `<span class="text-[10px] font-mono font-bold px-1.5 py-0.5 rounded bg-slate-800 text-brand-400 uppercase">ID: ${agency.idreal}</span>` : "";

                const popupContent = document.createElement("div");
                popupContent.className = "p-2 space-y-2 text-xs";
                popupContent.innerHTML = `
                    <div>
                        ${displayIdPopup}
                        <strong class="text-sm font-bold block text-white mt-1">${agency.nombre}</strong>
                        <p class="text-slate-400"><i class="fa fa-location-dot text-xs mr-1" style="color:${color}"></i>${agency.direccion || 'Sin dirección'}</p>
                    </div>
                    <div class="flex items-center gap-2">
                        <span class="px-2 py-0.5 rounded text-[10px] font-bold text-white uppercase" style="background-color: ${color}">
                            ${agency.zona}
                        </span>
                        <span class="px-2 py-0.5 rounded text-[10px] font-bold bg-slate-800 text-slate-300 uppercase">
                            ${agency.estado}
                        </span>
                    </div>
                    <button id="btnEdit-${agency.docId}" class="w-full mt-2 bg-brand-600 hover:bg-brand-500 text-white font-bold py-1.5 px-2 rounded-lg text-[11px] flex items-center justify-center gap-1.5 cursor-pointer">
                        <i class="fa fa-pen-to-square"></i> Editar Agencia
                    </button>
                `;

                marker.bindPopup(popupContent);

                marker.on('popupopen', () => {
                    const btnEdit = document.getElementById(`btnEdit-${agency.docId}`);
                    if (btnEdit) {
                        btnEdit.addEventListener('click', () => {
                            openEditModal(agency);
                        });
                    }
                });

                markersGroup.addLayer(marker);

                // Tarjeta Lateral
                const displayIdCard = agency.idreal ? `<p class="text-[10px] font-mono text-slate-500">ID: ${agency.idreal}</p>` : "";

                const card = document.createElement("div");
                card.className = "p-3 bg-slate-950/60 border border-slate-800 hover:border-brand-500/50 rounded-xl cursor-pointer transition-all space-y-1";
                card.innerHTML = `
                    <div class="flex justify-between items-center">
                        <h4 class="text-xs font-bold text-slate-200 truncate">${agency.nombre}</h4>
                        <span class="w-2.5 h-2.5 rounded-full shrink-0" style="background-color: ${color}"></span>
                    </div>
                    ${displayIdCard}
                    <p class="text-[10px] font-bold uppercase tracking-wider" style="color: ${color}">${agency.zona}</p>
                    <p class="text-[11px] text-slate-400 truncate">${agency.direccion || 'Sin dirección'}</p>
                `;

                card.addEventListener("click", () => {
                    map.flyTo([lat, lng], 16, { duration: 1.2 });
                    marker.openPopup();
                });

                agencyListContainer.appendChild(card);
            });

            if (bounds.length > 0) {
                map.fitBounds(bounds, { padding: [50, 50] });
            }
        }

        function openEditModal(agency) {
            editDocId.value = agency.docId;
            editNombre.value = agency.nombre;
            editZona.value = agency.zona;
            editEstado.value = agency.estado;
            editLat.value = agency.lat;
            editLng.value = agency.lng;
            editDireccion.value = agency.direccion;

            editAgencyModal.classList.remove("hidden");
        }

            e.preventDefault();
            adminEmail.value = auth.currentUser ? auth.currentUser.email : "";
            adminPassword.value = "";
            adminAuthModal.classList.remove("hidden");
        });

        adminAuthForm.addEventListener("submit", async (e) => {
            e.preventDefault();

            const email = adminEmail.value.trim();
            const pass = adminPassword.value;
            const btnAuth = document.getElementById("btnConfirmAuthAndSave");

            btnAuth.disabled = true;
            btnAuth.innerHTML = `<i class="fa fa-spinner fa-spin"></i> Verificando...`;

            try {
                await signInWithEmailAndPassword(auth, email, pass);

                btnAuth.innerHTML = `<i class="fa fa-spinner fa-spin"></i> Guardando cambios...`;
                
                const docId = editDocId.value;
                const newNombre = editNombre.value.trim();
                const newZona = editZona.value;
                const newEstado = editEstado.value;
                const newLat = parseFloat(editLat.value);
                const newLng = parseFloat(editLng.value);
                const newDireccion = editDireccion.value.trim();

                const agencyRef = doc(db, "agencias", docId);
                await updateDoc(agencyRef, {
                    nombre: newNombre,
                    zona: newZona,
                    estado: newEstado,
                    latitud: newLat,
                    longitud: newLng,
                    direccion: newDireccion,
                    updated_at: Date.now(),
                    searchable: `${newNombre} ${newZona} ${newDireccion}`.toLowerCase()
                });

                Toastify({ text: "Agencia actualizada correctamente con permisos de Admin", style: { background: "#10b981" } }).showToast();

                adminAuthModal.classList.add("hidden");
                editAgencyModal.classList.add("hidden");

                await loadData();

            } catch (err) {
                console.error("Error en re-autenticación o guardado:", err);
                Toastify({ text: "Credenciales inválidas o permisos insuficientes", style: { background: "#f43f5e" } }).showToast();
            } finally {
                btnAuth.disabled = false;
                btnAuth.innerHTML = `<i class="fa fa-check"></i> Autorizar y Guardar`;
            }
        });

        function applyFilters() {
            const selectedZone = zoneFilter.value;
            const searchText = mapSearchInput.value.toLowerCase().trim();

            const filtered = allAgencies.filter(agency => {
                const matchesZone = selectedZone === "ALL" || agency.zona === selectedZone;
                const matchesSearch = agency.nombre.toLowerCase().includes(searchText) || 
                                      (agency.idreal && agency.idreal.toLowerCase().includes(searchText)) ||
                                      agency.direccion.toLowerCase().includes(searchText);

                return matchesZone && matchesSearch;
            });

            renderAgenciesOnMap(filtered);
        }

        zoneFilter.addEventListener("change", applyFilters);
        mapSearchInput.addEventListener("input", applyFilters);

        document.getElementById("btnCloseModal").addEventListener("click", () => editAgencyModal.classList.add("hidden"));
        document.getElementById("btnCancelAgencyEdit").addEventListener("click", () => editAgencyModal.classList.add("hidden"));
        document.getElementById("btnCloseAuthModal").addEventListener("click", () => adminAuthModal.classList.add("hidden"));
        document.getElementById("btnCancelAuth").addEventListener("click", () => adminAuthModal.classList.add("hidden"));

        document.getElementById("btnLogout").addEventListener("click", async () => {
            await signOut(auth);
            window.location.href = "login.html";
        });

# Carpetas que se deben ignorar por completo
CARPETAS_IGNORADAS = {
    '.git', '.idea', '.vscode', 'node_modules', 'venv', 
    '__pycache__', 'build', 'dist', 'bin', 'obj'

# Archivos específicos que se deben ignorar

def concatenar_proyecto(directorio_raiz, archivo_salida):
    archivos_procesados = 0
    
    with open(archivo_salida, 'w', encoding='utf-8') as salida:
        salida.write(f"=== RESUMEN DEL PROYECTO ===\n")
        salida.write(f"Ruta base: {os.path.abspath(directorio_raiz)}\n\n")

        for raiz, directorios, archivos in os.walk(directorio_raiz):
            # Filtrar carpetas ignoradas en sitio para evitar recorrerlas
            directorios[:] = [d for d in directorios if d not in CARPETAS_IGNORADAS]

            for archivo in archivos:
                if archivo in ARCHIVOS_IGNORADOS:
                    continue

                _, ext = os.path.splitext(archivo)
                if ext.lower() in EXTENSIONES_PERMITIDAS:
                    ruta_completa = os.path.join(raiz, archivo)
                    ruta_relativa = os.path.relpath(ruta_completa, directorio_raiz)

                    salida.write("=" * 80 + "\n")
                    salida.write(f"ARCHIVO: {ruta_relativa}\n")
                    salida.write("=" * 80 + "\n\n")

                    try:
                        with open(ruta_completa, 'r', encoding='utf-8') as f:
                            salida.write(f.read())
                    except UnicodeDecodeError:
                        # Intentar con codificación fallback en caso de caracteres especiales
                        try:
                            with open(ruta_completa, 'r', encoding='latin-1') as f:
                                salida.write(f.read())
                        except Exception as e:
                            salida.write(f"[ERROR AL LEER ARCHIVO: {e}]\n")
                    except Exception as e:
                        salida.write(f"[ERROR AL LEER ARCHIVO: {e}]\n")

                    salida.write("\n\n")
                    archivos_procesados += 1

    print(f"✅ Proceso completado. Se unieron {archivos_procesados} archivos en '{archivo_salida}'.")


if __name__ == "__main__":
    # Ejecuta el script tomando como base la carpeta actual donde se ejecuta
    directorio_actual = os.getcwd()
    concatenar_proyecto(directorio_actual, ARCHIVO_SALIDA)

    const oldZ = oldZoneName.value.trim().toUpperCase();

    try {
        // 1. Crear nueva zona y borrar la antigua en el catálogo /zonas
        await setDoc(doc(db, "zonas", newZ), { nombre: newZ, updated_at: Date.now() });
        if (oldZ !== newZ) {
            await deleteDoc(doc(db, "zonas", oldZ));
        }

        // 2. Obtener TODAS las agencias para un barrido completo e impreciso (evita fallos de trim/case)
        const agenciasSnap = await getDocs(collection(db, "agencias"));
        const matchesToUpdate = [];

        agenciasSnap.forEach(docSnap => {
            const data = docSnap.data();
            if (data.zona) {
                // Normalización de texto para comparar
                const currentZonaNormalized = String(data.zona).trim().toUpperCase();
                if (currentZonaNormalized === oldZ) {
                    matchesToUpdate.push({ id: docSnap.id, data: data });
                }
            }
        });

        // 3. Actualizar en Firestore en lotes de 500 (Límite máximo de Batch en Firestore)
        if (matchesToUpdate.length > 0) {
            const BATCH_LIMIT = 500;
            for (let i = 0; i < matchesToUpdate.length; i += BATCH_LIMIT) {
                const chunk = matchesToUpdate.slice(i, i + BATCH_LIMIT);
                const batch = writeBatch(db);

                chunk.forEach(item => {
                    const ref = doc(db, "agencias", item.id);
                    const agencyData = item.data;

                    // Reconstruir campo searchable si aplica

                    const dir = agencyData.direccion || '';
                    const newSearchable = `${idReal} ${newZ} ${dir}`.toLowerCase();

                    batch.update(ref, {
                        zona: newZ,
                        updated_at: Date.now(),
                        searchable: newSearchable
                    });
                });

                await batch.commit();
            }
        }

        showToast(`Zona cambiada a "${newZ}" en ${matchesToUpdate.length} agencias`, "success");
        editModal.classList.add("hidden");

        btnConfirmEdit.innerHTML = `<i class="fa fa-arrows-rotate"></i> Actualizar en Todo`;
    }

// Protección estricta para Admin
export function requireAdmin() {
    onAuthStateChanged(auth, async (user) => {
        if (!user) {
            window.location.href = "login.html";
            return;
        }

        try {
            const userDocSnap = await getDoc(doc(db, "users", user.uid));
            if (userDocSnap.exists()) {
                const userData = userDocSnap.data();

                // Validar si es admin (por rol o username)
                const isAdmin = userData.role === "administrador" || userData.username === "admin";

                if (!isAdmin) {
                    alert("Acceso denegado. Se requieren permisos de Administrador.");
                    window.location.href = "index.html";
                }
            } else {
                window.location.href = "index.html";
            }
        } catch (error) {
            console.error("Error verificando rol:", error);
            window.location.href = "index.html";
        }
    });
}
